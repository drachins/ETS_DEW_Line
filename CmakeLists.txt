
cmake_minimum_required(VERSION 3.19)
project(real_time_transit)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CXX_FLAGS "-Wall -g -O0 -Q")
set(CMAKE_BUILD_TYPE Debug)
set(CMAKE_BUILD_TYPE RelWithDebInfo)

set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(Protobuf REQUIRED)
find_package(absl REQUIRED)
find_package(CURL REQUIRED)

include_directories(proto)
include_directories(${Protobuf_INCLUDE_DIR})
include_directories(${CMAKE_CURRENT_BINARY_DIR})

include_directories(${CURL_INCLUDE_DIRS})

add_executable(real_time_transit src/main.cpp src/trip.cpp src/realtimereader.cpp src/data_download.cpp proto/gtfs-realtime.pb.cc)

target_link_libraries(real_time_transit ${Protobuf_LIBRARIES} absl::log_internal_message absl::log_internal_check_op ${CURL_LIBRARIES})
